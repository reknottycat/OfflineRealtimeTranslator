cmake_minimum_required(VERSION 3.22.1)
project(lmt_runtime LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# Native inference must remain optimized even inside a debuggable APK.
set(CMAKE_BUILD_TYPE Release CACHE STRING "Native build type" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_OPENCL OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
add_subdirectory(../llama.cpp ${CMAKE_CURRENT_BINARY_DIR}/llama-build)
if(ANDROID)
    add_library(lmt_translator SHARED lmt_jni.cpp)
    target_link_libraries(lmt_translator PRIVATE llama log)
    # Hide static llama/ggml symbols from LiteRT-LM and other native libraries.
    target_link_options(lmt_translator PRIVATE "-Wl,--exclude-libs,ALL" "-Wl,-Bsymbolic" "-Wl,-z,max-page-size=16384")
else()
    add_executable(lmt-smoke lmt_smoke.cpp)
    target_link_libraries(lmt-smoke PRIVATE llama)
    add_executable(lmt-contract-tests lmt_contract_tests.cpp)
    target_link_libraries(lmt-contract-tests PRIVATE llama)
    add_executable(hymt2-smoke hymt2_smoke.cpp)
    target_link_libraries(hymt2-smoke PRIVATE llama)
    add_executable(hymt2-contract-tests hymt2_contract_tests.cpp)
    target_link_libraries(hymt2-contract-tests PRIVATE llama)
    enable_testing()
    add_test(NAME hymt2-contracts COMMAND hymt2-contract-tests)
    add_test(NAME lmt-contracts COMMAND lmt-contract-tests)
endif()
